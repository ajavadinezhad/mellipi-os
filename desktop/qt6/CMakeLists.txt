cmake_minimum_required(VERSION 3.21)
project(MellipiQt6 LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup()
qt_add_executable(mellipi-qt6 main.cpp)
qt_add_qml_module(mellipi-qt6 URI Mellipi VERSION 1.0 QML_FILES ui/Main.qml ui/Panel.qml ui/Launcher.qml)
target_link_libraries(mellipi-qt6 PRIVATE Qt6::Quick Qt6::QuickControls2)
enable_testing()
add_test(NAME qt6-first-frame COMMAND mellipi-qt6 --smoke)
set_tests_properties(qt6-first-frame PROPERTIES TIMEOUT 15
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
find_package(LayerShellQt CONFIG QUIET)
if(TARGET LayerShellQt::Interface)
    find_package(Qt6 REQUIRED COMPONENTS Network)
    target_link_libraries(mellipi-qt6 PRIVATE LayerShellQt::Interface Qt6::Network)
    target_compile_definitions(mellipi-qt6 PRIVATE MELLIPI_LAYER_SHELL)
endif()
